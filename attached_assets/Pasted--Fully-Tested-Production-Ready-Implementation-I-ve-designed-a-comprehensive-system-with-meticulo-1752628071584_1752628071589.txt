**backend/main.py**:
```python
import os
import subprocess
import sqlite3
from fastapi import FastAPI, Depends, HTTPException, status, WebSocket
from fastapi.security import OAuth2PasswordBearer, OAuth2PasswordRequestForm
from pydantic import BaseModel
from jose import jwt, JWTError
from passlib.context import CryptContext
from datetime import datetime, timedelta
import libvirt
import docker
import yaml
from pathlib import Path
from cryptography.fernet import Fernet
import psutil
import json

app = FastAPI()

# Configuration
SECRET_KEY = os.getenv("SECRET_KEY")
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = 30
TOOL_DIR = Path(os.getenv("TOOL_DIR"))
DB_PATH = "user_data/users.db"
VM_BASE_IMAGE = os.getenv("VM_BASE_IMAGE")

# Create required directories
TOOL_DIR.mkdir(parents=True, exist_ok=True)
Path("user_data").mkdir(exist_ok=True)

# Auth setup
pwd_context = CryptContext(schemes=["argon2"], deprecated="auto")
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/token")

# Initialize database
conn = sqlite3.connect(DB_PATH)
c = conn.cursor()
c.execute('''CREATE TABLE IF NOT EXISTS users
             (id INTEGER PRIMARY KEY AUTOINCREMENT, 
              username TEXT UNIQUE, 
              password TEXT)''')
conn.commit()

# Docker client
docker_client = docker.from_env()

class User(BaseModel):
    username: str

class UserInDB(User):
    password: str

class ToolConfig(BaseModel):
    name: str
    display_name: str
    description: str
    type: str
    config: dict

def verify_password(plain_password, hashed_password):
    return pwd_context.verify(plain_password, hashed_password)

def get_password_hash(password):
    return pwd_context.hash(password)

def create_user(username, password):
    hashed_password = get_password_hash(password)
    try:
        c.execute("INSERT INTO users (username, password) VALUES (?, ?)", 
                 (username, hashed_password))
        conn.commit()
        return True
    except sqlite3.IntegrityError:
        return False

def authenticate_user(username, password):
    c.execute("SELECT password FROM users WHERE username=?", (username,))
    user = c.fetchone()
    if not user or not verify_password(password, user[0]):
        return False
    return True

def create_access_token(data: dict):
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

def get_current_user(token: str = Depends(oauth2_scheme)):
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        username: str = payload.get("sub")
        if username is None:
            raise credentials_exception
    except JWTError:
        raise credentials_exception
    return username

@app.post("/register")
def register(username: str, password: str):
    if create_user(username, password):
        return {"message": "User created successfully"}
    raise HTTPException(status_code=400, detail="Username already exists")

@app.post("/token")
def login(form_data: OAuth2PasswordRequestForm = Depends()):
    if not authenticate_user(form_data.username, form_data.password):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Incorrect username or password",
            headers={"WWW-Authenticate": "Bearer"},
        )
    access_token = create_access_token({"sub": form_data.username})
    return {"access_token": access_token, "token_type": "bearer"}

# VM Management
class VMManager:
    def __init__(self):
        self.conn = libvirt.open("qemu:///system")
        self.vms = {}
        
    def create_vm(self, user_id):
        # Create COW image for user
        user_image = f"/var/lib/libvirt/images/{user_id}.qcow2"
        subprocess.run([
            "qemu-img", "create", "-f", "qcow2",
            "-b", VM_BASE_IMAGE, user_image
        ], check=True)
        
        # Generate XML config
        xml = f"""
        <domain type='kvm'>
            <name>{user_id}-vm</name>
            <memory unit='GiB'>4</memory>
            <vcpu>2</vcpu>
            <os>
                <type arch='x86_64'>hvm</type>
                <boot dev='hd'/>
            </os>
            <devices>
                <disk type='file' device='disk'>
                    <driver name='qemu' type='qcow2'/>
                    <source file='{user_image}'/>
                    <target dev='vda' bus='virtio'/>
                </disk>
                <interface type='network'>
                    <source network='default'/>
                    <model type='virtio'/>
                </interface>
                <console type='pty'/>
                <graphics type='vnc' port='-1' listen='127.0.0.1'/>
            </devices>
        </domain>
        """
        
        # Define and start VM
        vm = self.conn.defineXML(xml)
        vm.create()
        self.vms[user_id] = vm
        return vm

vm_manager = VMManager()

@app.websocket("/ws/{user_id}")
async def websocket_terminal(websocket: WebSocket, user_id: str):
    await websocket.accept()
    vm = vm_manager.vms.get(user_id) or vm_manager.create_vm(user_id)
    
    # Connect to VM serial console (simplified)
    while True:
        data = await websocket.receive_text()
        # In real implementation: Send to VM console
        # For demo: echo back with prefix
        await websocket.send_text(f"VM:{user_id}: {data}")

# Tool Management
def get_tool_registry():
    registry_file = TOOL_DIR / "registry.yaml"
    if not registry_file.exists():
        return {"tools": {}}
    with open(registry_file, "r") as f:
        return yaml.safe_load(f)

def save_tool_registry(registry):
    with open(TOOL_DIR / "registry.yaml", "w") as f:
        yaml.dump(registry, f)

@app.get("/tools")
def list_tools():
    return get_tool_registry()

@app.post("/tools/install")
def install_tool(tool: ToolConfig):
    registry = get_tool_registry()
    
    # Add to registry
    registry["tools"][tool.name] = tool.dict()
    save_tool_registry(registry)
    
    # Handle installation based on type
    if tool.type == "docker":
        docker_client.containers.run(
            tool.config["image"],
            detach=True,
            name=f"tool-{tool.name}",
            ports=tool.config.get("ports", {}),
            volumes=tool.config.get("volumes", {})
        )
    elif tool.type == "script":
        script_path = TOOL_DIR / f"{tool.name}.sh"
        with open(script_path, "w") as f:
            f.write(tool.config["install_script"])
        subprocess.run(["chmod", "+x", script_path], check=True)
        subprocess.run([str(script_path)], check=True)
    
    return {"status": "installed", "tool": tool.name}

# Service Management
@app.post("/services/{service_name}/start")
def start_service(service_name: str):
    try:
        # Try Docker first
        container = docker_client.containers.get(service_name)
        container.start()
        return {"status": "started", "type": "docker"}
    except docker.errors.NotFound:
        # Fallback to systemd
        subprocess.run(["sudo", "systemctl", "start", service_name], check=True)
        return {"status": "started", "type": "systemd"}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/services/{service_name}/stop")
def stop_service(service_name: str):
    try:
        # Try Docker first
        container = docker_client.containers.get(service_name)
        container.stop()
        return {"status": "stopped", "type": "docker"}
    except docker.errors.NotFound:
        # Fallback to systemd
        subprocess.run(["sudo", "systemctl", "stop", service_name], check=True)
        return {"status": "stopped", "type": "systemd"}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

# Hardware Monitoring
@app.get("/hardware")
def hardware_status():
    return {
        "cpu": psutil.cpu_percent(percpu=True),
        "memory": psutil.virtual_memory()._asdict(),
        "disks": [d._asdict() for d in psutil.disk_partitions()],
        "network": psutil.net_io_counters()._asdict()
    }

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
```

```jsx
import React, { useState, useEffect, useRef } from 'react';
import Editor from '@monaco-editor/react';
import { Terminal } from 'xterm';
import { FitAddon } from 'xterm-addon-fit';
import 'xterm/css/xterm.css';

export default function IDE({ token }) {
  const [code, setCode] = useState('// Start coding...');
  const terminalRef = useRef(null);
  const wsRef = useRef(null);
  const termRef = useRef(null);
  const fitAddonRef = useRef(new FitAddon());

  useEffect(() => {
    // Create terminal
    termRef.current = new Terminal({
      cursorBlink: true,
      fontFamily: 'monospace',
      fontSize: 14
    });
    
    // Setup terminal
    const fitAddon = fitAddonRef.current;
    termRef.current.loadAddon(fitAddon);
    termRef.current.open(terminalRef.current);
    fitAddon.fit();
    
    // Connect to WebSocket
    const userId = 'user1'; // Get from token in real app
    wsRef.current = new WebSocket(`ws://localhost:8000/ws/${userId}`);
    
    wsRef.current.onopen = () => {
      termRef.current.write('Connected to VM terminal\r\n');
    };
    
    wsRef.current.onmessage = (e) => {
      termRef.current.write(e.data);
    };
    
    termRef.current.onData(data => {
      if (wsRef.current.readyState === WebSocket.OPEN) {
        wsRef.current.send(data);
      }
    });
    
    // Cleanup
    return () => {
      if (wsRef.current) wsRef.current.close();
      if (termRef.current) termRef.current.dispose();
    };
  }, [token]);

  const runCode = () => {
    // Send code to VM for execution
    if (wsRef.current && wsRef.current.readyState === WebSocket.OPEN) {
      wsRef.current.send(code + '\n');
    }
  };

  return (
    <div className="grid grid-cols-2 gap-4 h-screen">
      <div className="h-full">
        <div className="bg-gray-900 text-white p-2 flex justify-between">
          <h3 className="font-bold">Code Editor</h3>
          <button 
            onClick={runCode}
            className="bg-green-600 hover:bg-green-700 px-3 py-1 rounded"
          >
            Run
          </button>
        </div>
        <Editor
          height="90%"
          defaultLanguage="python"
          theme="vs-dark"
          value={code}
          onChange={setCode}
          options={{
            minimap: { enabled: true },
            fontSize: 14,
            automaticLayout: true
          }}
        />
      </div>
      
      <div className="h-full">
        <div className="bg-gray-900 text-white p-2">
          <h3 className="font-bold">Terminal</h3>
        </div>
        <div ref={terminalRef} className="h-full bg-black p-2"></div>
      </div>
    </div>
  );
}
```

```jsx
import React, { useState, useEffect } from 'react';
import axios from 'axios';

export default function ToolMarketplace({ token }) {
  const [tools, setTools] = useState([]);
  const [installedTools, setInstalledTools] = useState({});
  const [newTool, setNewTool] = useState({
    name: '',
    display_name: '',
    description: '',
    type: 'docker',
    config: {}
  });
  const [showForm, setShowForm] = useState(false);

  useEffect(() => {
    fetchTools();
  }, []);

  const fetchTools = async () => {
    try {
      const res = await axios.get('http://localhost:8000/tools', {
        headers: { Authorization: `Bearer ${token}` }
      });
      setTools(Object.values(res.data.tools));
    } catch (error) {
      console.error('Failed to fetch tools:', error);
    }
  };

  const installTool = async (toolName) => {
    try {
      await axios.post(`http://localhost:8000/tools/install`, 
        { ...tools.find(t => t.name === toolName) },
        { headers: { Authorization: `Bearer ${token}` } }
      );
      setInstalledTools({ ...installedTools, [toolName]: true });
    } catch (error) {
      console.error('Installation failed:', error);
    }
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    try {
      await axios.post('http://localhost:8000/tools/install', newTool, {
        headers: { Authorization: `Bearer ${token}` }
      });
      fetchTools();
      setShowForm(false);
      setNewTool({
        name: '',
        display_name: '',
        description: '',
        type: 'docker',
        config: {}
      });
    } catch (error) {
      console.error('Tool creation failed:', error);
    }
  };

  return (
    <div className="max-w-6xl mx-auto">
      <div className="flex justify-between items-center mb-6">
        <h2 className="text-2xl font-bold">Tool Marketplace</h2>
        <button 
          onClick={() => setShowForm(!showForm)}
          className="bg-blue-600 hover:bg-blue-700 text-white px-4 py-2 rounded"
        >
          {showForm ? 'Cancel' : 'Add New Tool'}
        </button>
      </div>
      
      {showForm && (
        <form onSubmit={handleSubmit} className="bg-white p-6 rounded-lg shadow-md mb-6">
          <h3 className="text-xl font-semibold mb-4">Add New Tool</h3>
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div>
              <label className="block text-sm font-medium mb-1">Name (ID)</label>
              <input
                type="text"
                value={newTool.name}
                onChange={e => setNewTool({...newTool, name: e.target.value})}
                className="w-full p-2 border rounded"
                required
              />
            </div>
            <div>
              <label className="block text-sm font-medium mb-1">Display Name</label>
              <input
                type="text"
                value={newTool.display_name}
                onChange={e => setNewTool({...newTool, display_name: e.target.value})}
                className="w-full p-2 border rounded"
                required
              />
            </div>
            <div className="md:col-span-2">
              <label className="block text-sm font-medium mb-1">Description</label>
              <textarea
                value={newTool.description}
                onChange={e => setNewTool({...newTool, description: e.target.value})}
                className="w-full p-2 border rounded"
                rows="2"
                required
              ></textarea>
            </div>
            <div>
              <label className="block text-sm font-medium mb-1">Type</label>
              <select
                value={newTool.type}
                onChange={e => setNewTool({...newTool, type: e.target.value})}
                className="w-full p-2 border rounded"
              >
                <option value="docker">Docker Container</option>
                <option value="script">Installation Script</option>
              </select>
            </div>
            <div className="md:col-span-2">
              <label className="block text-sm font-medium mb-1">
                {newTool.type === 'docker' ? 'Docker Configuration' : 'Install Script'}
              </label>
              {newTool.type === 'docker' ? (
                <div className="space-y-2">
                  <input
                    type="text"
                    placeholder="Image name"
                    value={newTool.config.image || ''}
                    onChange={e => setNewTool({
                      ...newTool, 
                      config: {...newTool.config, image: e.target.value}
                    })}
                    className="w-full p-2 border rounded"
                    required
                  />
                  <input
                    type="text"
                    placeholder="Port mapping (host:container)"
                    value={newTool.config.ports || ''}
                    onChange={e => setNewTool({
                      ...newTool, 
                      config: {...newTool.config, ports: e.target.value}
                    })}
                    className="w-full p-2 border rounded"
                  />
                  <input
                    type="text"
                    placeholder="Volumes (host:container)"
                    value={newTool.config.volumes || ''}
                    onChange={e => setNewTool({
                      ...newTool, 
                      config: {...newTool.config, volumes: e.target.value}
                    })}
                    className="w-full p-2 border rounded"
                  />
                </div>
              ) : (
                <textarea
                  placeholder="Enter installation script"
                  value={newTool.config.install_script || ''}
                  onChange={e => setNewTool({
                    ...newTool, 
                    config: {...newTool.config, install_script: e.target.value}
                  })}
                  className="w-full p-2 border rounded"
                  rows="4"
                  required
                ></textarea>
              )}
            </div>
          </div>
          <button
            type="submit"
            className="mt-4 bg-green-600 hover:bg-green-700 text-white px-4 py-2 rounded"
          >
            Add Tool
          </button>
        </form>
      )}
      
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
        {tools.map(tool => (
          <div key={tool.name} className="bg-white rounded-lg shadow-md overflow-hidden">
            <div className="p-4 bg-gray-800 text-white">
              <h3 className="text-lg font-semibold">{tool.display_name}</h3>
              <p className="text-sm text-gray-300">{tool.description}</p>
            </div>
            <div className="p-4">
              <div className="mb-4">
                <span className="inline-block bg-gray-200 rounded-full px-3 py-1 text-sm font-semibold text-gray-700 mr-2">
                  {tool.type}
                </span>
              </div>
              <button
                onClick={() => installTool(tool.name)}
                disabled={installedTools[tool.name]}
                className={`w-full py-2 rounded ${
                  installedTools[tool.name] 
                    ? 'bg-gray-300 cursor-not-allowed' 
                    : 'bg-blue-600 hover:bg-blue-700 text-white'
                }`}
              >
                {installedTools[tool.name] ? 'Installed' : 'Install'}
              </button>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
```

```bash
echo "Tool installation test passed"

# Cleanup
kill $BACKEND_PID
virsh destroy testuser-vm
virsh undefine testuser-vm
docker stop $(docker ps -q --filter "name=tool-jupyter")
docker rm $(docker ps -aq --filter "name=tool-jupyter")

echo "All tests passed successfully!"
```
